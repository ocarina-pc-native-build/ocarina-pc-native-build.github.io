cmake_minimum_required(VERSION 3.24)
project(OoT2026_PC_Recomp VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This runtime currently targets Windows x64")
endif()
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Use Clang/LLVM; upstream generated C uses GNU extensions")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(OOT_DEPS "" CACHE PATH "Directory prepared by tools/bootstrap.py")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_path(SDL3_INCLUDE SDL3/SDL.h HINTS "${OOT_DEPS}/sdl3-devel/SDL3-3.4.18/x86_64-w64-mingw32/include" REQUIRED)
find_library(SDL3_LIB NAMES SDL3 HINTS "${OOT_DEPS}/sdl3-devel/SDL3-3.4.18/x86_64-w64-mingw32/lib" REQUIRED)
find_path(VK_INCLUDE vulkan/vulkan.h HINTS "${OOT_DEPS}/vulkan-headers/ucrt64/include" REQUIRED)
set(SDL3_DLL "${OOT_DEPS}/sdl3-devel/SDL3-3.4.18/x86_64-w64-mingw32/bin/SDL3.dll")
add_library(oot_host STATIC src/host.c src/gpu.c)
target_include_directories(oot_host PUBLIC src PRIVATE "${SDL3_INCLUDE}" "${VK_INCLUDE}")
target_compile_definitions(oot_host PUBLIC _WIN32_WINNT=0x0A00)
target_compile_options(oot_host PRIVATE -Wall -Wextra)
target_link_libraries(oot_host PUBLIC "${SDL3_LIB}")
add_executable(oot-runtime src/main.cpp)
target_link_libraries(oot-runtime PRIVATE oot_host shell32 ole32)
target_link_options(oot-runtime PRIVATE -static)
add_custom_command(TARGET oot-runtime POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SDL3_DLL}" "$<TARGET_FILE_DIR:oot-runtime>")
set(LIFTED "${CMAKE_CURRENT_BINARY_DIR}/synthetic.c")
add_custom_command(OUTPUT "${LIFTED}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/synthetic.py" "${LIFTED}" DEPENDS tools/synthetic.py third_party/dkb/recomp/lifter.py third_party/dkb/recomp/simd_c.py)
add_library(oot_lifted_test SHARED "${LIFTED}")
target_include_directories(oot_lifted_test PRIVATE third_party/dkb/runtime)
target_compile_options(oot_lifted_test PRIVATE -march=x86-64-v3 -fno-strict-aliasing)
target_link_options(oot_lifted_test PRIVATE -static)
set_target_properties(oot_lifted_test PROPERTIES PREFIX "")
enable_testing()
set(PACGA_LIFTED "${CMAKE_CURRENT_BINARY_DIR}/pacga_synthetic.c")
add_custom_command(OUTPUT "${PACGA_LIFTED}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/test_pacga.py" --generate "${PACGA_LIFTED}" DEPENDS tools/test_pacga.py third_party/dkb/recomp/lifter.py)
add_library(oot_pacga_test SHARED third_party/dkb/runtime/pacga.c "${PACGA_LIFTED}")
target_include_directories(oot_pacga_test PRIVATE third_party/dkb/runtime)
target_compile_options(oot_pacga_test PRIVATE -march=x86-64-v3)
target_compile_definitions(oot_pacga_test PRIVATE OOT_PACGA_TEST)
target_link_libraries(oot_pacga_test PRIVATE bcrypt)
target_link_options(oot_pacga_test PRIVATE -static)
set_target_properties(oot_pacga_test PROPERTIES PREFIX "")
add_test(NAME pacga_differential COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/test_pacga.py" "$<TARGET_FILE:oot_pacga_test>")
add_test(NAME host_selftest COMMAND oot-runtime --self-test)
add_test(NAME invalid_nso COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_SOURCE_DIR}/tests" -v)
add_test(NAME differential_aarch64 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/synthetic.py" --test "$<TARGET_FILE:oot_lifted_test>")
install(TARGETS oot-runtime RUNTIME DESTINATION bin)
install(FILES "${SDL3_DLL}" DESTINATION bin)
